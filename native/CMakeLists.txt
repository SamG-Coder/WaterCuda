cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
project(WaterCudaNative LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_executable(watercuda src/main.cpp src/renderer.cu)
target_include_directories(watercuda PRIVATE src)
target_compile_definitions(watercuda PRIVATE WATER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(watercuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
if(WIN32)
  target_compile_definitions(watercuda PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(watercuda PRIVATE user32 gdi32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME cuda-parity COMMAND watercuda --self-test)
  add_test(NAME render-smoke COMMAND watercuda --headless --width 320 --height 192 --frames 3 --seed 884 --view bars --output smoke.bmp)
  if(WIN32)
    add_test(NAME window-smoke COMMAND watercuda --hidden --frames 3 --width 640 --height 360 --view scrub --output window-smoke.bmp)
  endif()
  add_test(NAME underwater-smoke COMMAND watercuda --headless --width 320 --height 192 --frames 3 --seed 884 --view reef --output reef-smoke.bmp)
  add_test(NAME weather-smoke COMMAND watercuda --headless --width 320 --height 192 --frames 3 --seed 884 --view shore --weather storm)
  add_test(NAME shrub-smoke COMMAND watercuda --headless --width 320 --height 192 --frames 2 --seed 884 --view scrub)
endif()
